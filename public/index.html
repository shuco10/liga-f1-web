// FORMULARIOS DE ENVÍO - ACTUALIZADOS PARA REFRESCAR AL INSTANTE document.getElementById('form-piloto').addEventListener('submit', async (e) => { e.preventDefault(); const datos = { gamertag: document.getElementById('gamertag').value, plataforma: document.getElementById('plataforma').value, escuderia_id: parseInt(document.getElementById('escuderia_id').value) }; const res = await fetch('/api/nuevo-piloto', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(datos) }); if(res.ok) { document.getElementById('msg-piloto').innerText = "¡Piloto inscrito con éxito!"; document.getElementById('form-piloto').reset(); await cargarTodosLosDatos(); // Fuerza a las tablas a actualizarse await cargarDesplegablesAdmin(); // Actualiza los select de carreras y sanciones } else { document.getElementById('msg-piloto').innerText = "Error al guardar en el servidor."; } }); document.getElementById('form-carrera').addEventListener('submit', async (e) => { e.preventDefault(); const datos = { piloto_id: parseInt(document.getElementById('select-piloto').value), posicion_carrera: parseInt(document.getElementById('posicion_carrera').value) }; const res = await fetch('/api/subir-resultado', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(datos) }); if(res.ok) { document.getElementById('msg-carrera').innerText = "¡Puntos sumados correctamente!"; document.getElementById('form-carrera').reset(); await cargarTodosLosDatos(); // Actualiza la clasificación al instante } }); document.getElementById('form-sancion').addEventListener('submit', async (e) => { e.preventDefault(); const datos = { piloto_id: parseInt(document.getElementById('select-piloto-sancion').value), puntos_restar: parseInt(document.getElementById('puntos_restar').value) }; const res = await fetch('/api/restar-licencia', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(datos) }); if(res.ok) { document.getElementById('msg-sancion').innerText = "¡Sanción aplicada al carnet!"; document.getElementById('form-sancion').reset(); await cargarTodosLosDatos(); // Actualiza la tabla de sanciones al instante } });
